cmake_minimum_required(VERSION 3.13)

# Default RP2040 Pico; override with -DPICO_BOARD=pico_w (docker/Makefile PICO_BOARD).
set(PICO_BOARD pico CACHE STRING "RP2040 board type")

# Include the standard Raspberry Pi Pico SDK setup (PICO_SDK_PATH from pico-dev).
include(pico_sdk_import.cmake)

project(crt_drive C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Initialize the SDK
pico_sdk_init()

# Shared 78 Hz 80-col scanout (PIO + DMA + packing).
add_library(crt_video video/video.c)
pico_generate_pio_header(crt_video ${CMAKE_CURRENT_LIST_DIR}/video/video_pixel.pio)
pico_generate_pio_header(crt_video ${CMAKE_CURRENT_LIST_DIR}/video/hsync.pio)
pico_generate_pio_header(crt_video ${CMAKE_CURRENT_LIST_DIR}/video/vsync.pio)
target_include_directories(crt_video PUBLIC
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_LIST_DIR}/video
)
target_link_libraries(crt_video PUBLIC
    pico_stdlib
    hardware_pio
    hardware_dma
    hardware_clocks
    hardware_irq
)

# Analog-setup patterns (service tool).
add_executable(crt_drive
    pattern/main.c
)
target_link_libraries(crt_drive crt_video)
pico_enable_stdio_usb(crt_drive 1)
pico_enable_stdio_uart(crt_drive 0)
pico_add_extra_outputs(crt_drive)

# Glass TTY (terminal emulator core).
set(TERM_IMAGE_ID "" CACHE STRING "Unique crt_term CDC digest")
set(_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
if(TERM_IMAGE_ID)
  # keep -DTERM_IMAGE_ID / cache
elseif(DEFINED ENV{TERM_IMAGE_ID} AND NOT "$ENV{TERM_IMAGE_ID}" STREQUAL "")
  set(TERM_IMAGE_ID "$ENV{TERM_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE TERM_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT TERM_IMAGE_ID)
  set(TERM_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^A-Za-z0-9._-]" "" TERM_IMAGE_ID "${TERM_IMAGE_ID}")
if(TERM_IMAGE_ID STREQUAL "")
  set(TERM_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/term_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef TERM_BUILD_ID_H\n"
  "#define TERM_BUILD_ID_H\n"
  "#define TERM_IMAGE_ID \"${TERM_IMAGE_ID}\"\n"
  "#endif\n"
)

add_executable(crt_term
    term/main.c
    term/screen.c
    term/font_7x10.c
    term/tty.c
)
target_include_directories(crt_term PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}/term
    ${CMAKE_CURRENT_BINARY_DIR}
)
target_link_libraries(crt_term crt_video)
target_compile_definitions(crt_term PRIVATE PICO_STDIO_USB_CONNECT_WAIT_MS=5000)
pico_enable_stdio_usb(crt_term 1)
pico_enable_stdio_uart(crt_term 0)
pico_add_extra_outputs(crt_term)
