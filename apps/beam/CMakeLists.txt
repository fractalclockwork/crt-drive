cmake_minimum_required(VERSION 3.13)

# Beam-race tests. Which scan line DMA is painting, and four ways to write
# the framebuffer against that line.
#   cmake -S apps/beam -B build/beam
# Or from the repo root: make build APP=beam

set(PICO_BOARD pico CACHE STRING "RP2040 board type")
get_filename_component(CRT_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${CRT_REPO_ROOT}/pico_sdk_import.cmake")

project(crt_beam C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

set(BEAM_IMAGE_ID "" CACHE STRING "Unique crt_beam CDC digest")
if(BEAM_IMAGE_ID)
  # keep -DBEAM_IMAGE_ID / cache
elseif(DEFINED ENV{BEAM_IMAGE_ID} AND NOT "$ENV{BEAM_IMAGE_ID}" STREQUAL "")
  set(BEAM_IMAGE_ID "$ENV{BEAM_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${CRT_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE BEAM_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT BEAM_IMAGE_ID)
  set(BEAM_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^-A-Za-z0-9._]" "" BEAM_IMAGE_ID "${BEAM_IMAGE_ID}")
if(BEAM_IMAGE_ID STREQUAL "")
  set(BEAM_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/beam_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef BEAM_BUILD_ID_H\n"
  "#define BEAM_BUILD_ID_H\n"
  "#define BEAM_IMAGE_ID \"${BEAM_IMAGE_ID}\"\n"
  "#endif\n"
)

add_executable(crt_beam
    main.c
    "${CRT_REPO_ROOT}/video/scanout.c"
)
pico_generate_pio_header(crt_beam "${CRT_REPO_ROOT}/video/video_pixel.pio")
pico_generate_pio_header(crt_beam "${CRT_REPO_ROOT}/video/hsync60.pio")
pico_generate_pio_header(crt_beam "${CRT_REPO_ROOT}/video/hsync60_132.pio")
pico_generate_pio_header(crt_beam "${CRT_REPO_ROOT}/video/vsync60.pio")
pico_generate_pio_header(crt_beam "${CRT_REPO_ROOT}/video/vsync.pio")
target_include_directories(crt_beam PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_BINARY_DIR}
    "${CRT_REPO_ROOT}"
    "${CRT_REPO_ROOT}/video"
)
target_link_libraries(crt_beam
    pico_stdlib
    hardware_pio
    hardware_dma
    hardware_clocks
    hardware_irq
)
target_compile_definitions(crt_beam PRIVATE PICO_STDIO_USB_CONNECT_WAIT_MS=5000)
pico_enable_stdio_usb(crt_beam 1)
pico_enable_stdio_uart(crt_beam 0)
pico_add_extra_outputs(crt_beam)
