cmake_minimum_required(VERSION 3.13)

# Phosphor physics attract reel at all four Appendix B rates.
#   cmake -S apps/demos -B build/demos
# Or from the repo root: make build APP=demos

set(PICO_BOARD pico CACHE STRING "RP2040 board type")
get_filename_component(CRT_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${CRT_REPO_ROOT}/pico_sdk_import.cmake")

project(crt_demos C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

set(DEMO_IMAGE_ID "" CACHE STRING "Unique crt_demos CDC digest")
if(DEMO_IMAGE_ID)
  # keep -DDEMO_IMAGE_ID / cache
elseif(DEFINED ENV{DEMO_IMAGE_ID} AND NOT "$ENV{DEMO_IMAGE_ID}" STREQUAL "")
  set(DEMO_IMAGE_ID "$ENV{DEMO_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${CRT_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE DEMO_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT DEMO_IMAGE_ID)
  set(DEMO_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^-A-Za-z0-9._]" "" DEMO_IMAGE_ID "${DEMO_IMAGE_ID}")
if(DEMO_IMAGE_ID STREQUAL "")
  set(DEMO_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/demo_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef DEMO_BUILD_ID_H\n"
  "#define DEMO_BUILD_ID_H\n"
  "#define DEMO_IMAGE_ID \"${DEMO_IMAGE_ID}\"\n"
  "#endif\n"
)

add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
  COMMAND python3 "${CRT_REPO_ROOT}/tools/ttf_to_c.py"
          "${CRT_REPO_ROOT}/font/fonts/NotoSans-Regular.ttf"
          "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
          font_noto_sans_regular
  DEPENDS
    "${CRT_REPO_ROOT}/tools/ttf_to_c.py"
    "${CRT_REPO_ROOT}/font/fonts/NotoSans-Regular.ttf"
  VERBATIM
)

add_executable(crt_demos
    main.c
    gfx.c
    scenes.c
    "${CRT_REPO_ROOT}/video/scanout.c"
    "${CRT_REPO_ROOT}/font/font.c"
    "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
)
pico_generate_pio_header(crt_demos "${CRT_REPO_ROOT}/video/video_pixel.pio")
pico_generate_pio_header(crt_demos "${CRT_REPO_ROOT}/video/hsync60.pio")
pico_generate_pio_header(crt_demos "${CRT_REPO_ROOT}/video/hsync60_132.pio")
pico_generate_pio_header(crt_demos "${CRT_REPO_ROOT}/video/vsync60.pio")
pico_generate_pio_header(crt_demos "${CRT_REPO_ROOT}/video/vsync.pio")
target_include_directories(crt_demos PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_BINARY_DIR}
    "${CRT_REPO_ROOT}"
    "${CRT_REPO_ROOT}/video"
    "${CRT_REPO_ROOT}/font"
)
target_link_libraries(crt_demos
    pico_stdlib
    hardware_pio
    hardware_dma
    hardware_clocks
    hardware_irq
    hardware_sync
)
target_compile_definitions(crt_demos PRIVATE PICO_STDIO_USB_CONNECT_WAIT_MS=5000)
pico_enable_stdio_usb(crt_demos 1)
pico_enable_stdio_uart(crt_demos 0)
pico_add_extra_outputs(crt_demos)
