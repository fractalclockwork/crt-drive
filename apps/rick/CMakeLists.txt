cmake_minimum_required(VERSION 3.13)

# Rickroll. Packed 2 bpp frames in flash, pixel DMA reads them directly.
#   cmake -S apps/rick -B build/rick
# Or from the repo root: make build APP=rick
# The frame pack is produced on the Dev-Host before cmake:
#   python3 tools/pack_rick.py apps/rick/rick_frames.pack apps/rick/frames.h

set(PICO_BOARD pico CACHE STRING "RP2040 board type")
get_filename_component(CRT_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${CRT_REPO_ROOT}/pico_sdk_import.cmake")

project(crt_rick C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

set(RICK_IMAGE_ID "" CACHE STRING "Unique crt_rick CDC digest")
if(RICK_IMAGE_ID)
  # keep -DRICK_IMAGE_ID / cache
elseif(DEFINED ENV{RICK_IMAGE_ID} AND NOT "$ENV{RICK_IMAGE_ID}" STREQUAL "")
  set(RICK_IMAGE_ID "$ENV{RICK_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${CRT_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE RICK_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT RICK_IMAGE_ID)
  set(RICK_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^-A-Za-z0-9._]" "" RICK_IMAGE_ID "${RICK_IMAGE_ID}")
if(RICK_IMAGE_ID STREQUAL "")
  set(RICK_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/rick_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef RICK_BUILD_ID_H\n"
  "#define RICK_BUILD_ID_H\n"
  "#define RICK_IMAGE_ID \"${RICK_IMAGE_ID}\"\n"
  "#endif\n"
)

set(RICK_PACK "${CMAKE_CURRENT_LIST_DIR}/rick_frames.pack")
if(NOT EXISTS "${RICK_PACK}")
  message(FATAL_ERROR "missing ${RICK_PACK} — python3 tools/pack_rick.py apps/rick/rick_frames.pack apps/rick/frames.h")
endif()

# objcopy names symbols from the file name, so the input must be
# rick_frames.pack and not an absolute path. .rodata stays in flash;
# a .data blob would be copied into SRAM and would not fit.
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rick_frames.o"
  COMMAND ${CMAKE_COMMAND} -E chdir "${CMAKE_CURRENT_LIST_DIR}"
          ${CMAKE_OBJCOPY} -I binary -O elf32-littlearm -B arm
          --set-section-alignment .data=4
          --rename-section .data=.rodata,alloc,load,readonly,data,contents
          rick_frames.pack "${CMAKE_CURRENT_BINARY_DIR}/rick_frames.o"
  DEPENDS "${RICK_PACK}"
  VERBATIM
)

add_executable(crt_rick
    main.c
    "${CMAKE_CURRENT_BINARY_DIR}/rick_frames.o"
    "${CRT_REPO_ROOT}/video/scanout.c"
)
pico_generate_pio_header(crt_rick "${CRT_REPO_ROOT}/video/video_pixel.pio")
pico_generate_pio_header(crt_rick "${CRT_REPO_ROOT}/video/hsync60.pio")
pico_generate_pio_header(crt_rick "${CRT_REPO_ROOT}/video/hsync60_132.pio")
pico_generate_pio_header(crt_rick "${CRT_REPO_ROOT}/video/vsync60.pio")
pico_generate_pio_header(crt_rick "${CRT_REPO_ROOT}/video/vsync.pio")
target_include_directories(crt_rick PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_BINARY_DIR}
    "${CRT_REPO_ROOT}"
    "${CRT_REPO_ROOT}/video"
)
target_link_libraries(crt_rick
    pico_stdlib
    hardware_pio
    hardware_dma
    hardware_clocks
    hardware_irq
    hardware_sync
)
target_link_options(crt_rick PRIVATE -Wl,-u,_binary_rick_frames_pack_start)
target_compile_definitions(crt_rick PRIVATE PICO_STDIO_USB_CONNECT_WAIT_MS=5000)
pico_enable_stdio_usb(crt_rick 1)
pico_enable_stdio_uart(crt_rick 0)
pico_add_extra_outputs(crt_rick)
