cmake_minimum_required(VERSION 3.13)

# Video TTY. Text on USB, pictures from flash.
#   cmake -S apps/vtty -B build/vtty
# Or from the repo root: make build APP=vtty

set(PICO_BOARD pico CACHE STRING "RP2040 board type")
get_filename_component(CRT_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${CRT_REPO_ROOT}/pico_sdk_import.cmake")

project(crt_vtty C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Glyph rasterization recurses while flattening curves. The default 2 KB
# stack is the call chain plus the 512-byte scanline.
set(PICO_STACK_SIZE 0x2000 CACHE STRING "core 0 stack for glyph rasterization" FORCE)

pico_sdk_init()

set(VTTY_IMAGE_ID "" CACHE STRING "Unique crt_vtty CDC digest")
if(VTTY_IMAGE_ID)
  # keep -DVTTY_IMAGE_ID / cache
elseif(DEFINED ENV{VTTY_IMAGE_ID} AND NOT "$ENV{VTTY_IMAGE_ID}" STREQUAL "")
  set(VTTY_IMAGE_ID "$ENV{VTTY_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${CRT_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE VTTY_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT VTTY_IMAGE_ID)
  set(VTTY_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^-A-Za-z0-9._]" "" VTTY_IMAGE_ID "${VTTY_IMAGE_ID}")
if(VTTY_IMAGE_ID STREQUAL "")
  set(VTTY_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/vtty_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef VTTY_BUILD_ID_H\n"
  "#define VTTY_BUILD_ID_H\n"
  "#define VTTY_IMAGE_ID \"${VTTY_IMAGE_ID}\"\n"
  "#endif\n"
)

add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
  COMMAND python3 "${CRT_REPO_ROOT}/tools/ttf_to_c.py"
          "${CRT_REPO_ROOT}/font/fonts/NotoSans-Regular.ttf"
          "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
          font_noto_sans_regular
  DEPENDS
    "${CRT_REPO_ROOT}/tools/ttf_to_c.py"
    "${CRT_REPO_ROOT}/font/fonts/NotoSans-Regular.ttf"
  VERBATIM
)

add_executable(crt_vtty
    main.c
    recover.c
    session.c
    catalog.c
    builtin.c
    "${CRT_REPO_ROOT}/apps/term/screen.c"
    "${CRT_REPO_ROOT}/apps/term/font_7x10.c"
    "${CRT_REPO_ROOT}/apps/term/tty.c"
    "${CRT_REPO_ROOT}/video/indian_head.c"
    "${CRT_REPO_ROOT}/video/scanout.c"
    "${CRT_REPO_ROOT}/font/font.c"
    "${CMAKE_CURRENT_BINARY_DIR}/noto_sans_regular.c"
)
pico_generate_pio_header(crt_vtty "${CRT_REPO_ROOT}/video/video_pixel.pio")
pico_generate_pio_header(crt_vtty "${CRT_REPO_ROOT}/video/hsync60.pio")
pico_generate_pio_header(crt_vtty "${CRT_REPO_ROOT}/video/hsync60_132.pio")
pico_generate_pio_header(crt_vtty "${CRT_REPO_ROOT}/video/vsync60.pio")
pico_generate_pio_header(crt_vtty "${CRT_REPO_ROOT}/video/vsync.pio")
target_include_directories(crt_vtty PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_BINARY_DIR}
    "${CRT_REPO_ROOT}/apps/term"
    "${CRT_REPO_ROOT}/video"
)
target_link_libraries(crt_vtty
    pico_stdlib
    hardware_pio
    hardware_dma
    hardware_clocks
    hardware_irq
    hardware_flash
    hardware_watchdog
)
target_compile_definitions(crt_vtty PRIVATE
    PICO_STDIO_USB_CONNECT_WAIT_MS=5000
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=100000
)
pico_enable_stdio_usb(crt_vtty 1)
pico_enable_stdio_uart(crt_vtty 0)
pico_add_extra_outputs(crt_vtty)
