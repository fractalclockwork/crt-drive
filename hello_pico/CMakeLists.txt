cmake_minimum_required(VERSION 3.13)

# Dev-Host USB bring-up firmware (not the CRT PIO drive).
set(PICO_BOARD pico CACHE STRING "RP2040 board type")

include("${CMAKE_CURRENT_LIST_DIR}/../pico_sdk_import.cmake")

project(hello_pico C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

set(_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(DEFINED ENV{HELLO_IMAGE_ID} AND NOT "$ENV{HELLO_IMAGE_ID}" STREQUAL "")
  set(HELLO_IMAGE_ID "$ENV{HELLO_IMAGE_ID}")
else()
  execute_process(
    COMMAND git -C "${_REPO_ROOT}" rev-parse --short=12 HEAD
    OUTPUT_VARIABLE HELLO_IMAGE_ID
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
if(NOT HELLO_IMAGE_ID)
  set(HELLO_IMAGE_ID "unknown")
endif()
string(REGEX REPLACE "[^A-Za-z0-9._-]" "" HELLO_IMAGE_ID "${HELLO_IMAGE_ID}")
if(HELLO_IMAGE_ID STREQUAL "")
  set(HELLO_IMAGE_ID "unknown")
endif()

file(
  WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello_build_id.h"
  "/* Generated by CMake — do not edit. */\n"
  "#ifndef HELLO_BUILD_ID_H\n"
  "#define HELLO_BUILD_ID_H\n"
  "#define HELLO_IMAGE_ID \"${HELLO_IMAGE_ID}\"\n"
  "#endif\n"
)

add_executable(hello_pico src/main.c)
target_include_directories(hello_pico PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(hello_pico pico_stdlib)
target_compile_definitions(hello_pico PRIVATE PICO_STDIO_USB_CONNECT_WAIT_MS=5000)

pico_enable_stdio_usb(hello_pico 1)
pico_enable_stdio_uart(hello_pico 0)
pico_add_extra_outputs(hello_pico)
